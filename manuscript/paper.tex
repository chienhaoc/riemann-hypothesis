\begin{abstract}
This paper presents a comprehensive, rigorous functional-analytic framework for the Riemann Hypothesis (RH) centered on André Weil's positivity criterion, Suzuki's continuous screw-kernel representation, and de Branges--Herglotz canonical Hamiltonian systems. We analyze the algebraic structure of the arithmetic explicit formula on positive-type convolution spaces, elucidating the role of the von Mangoldt function $\Lambda(n) \ge 0$ in Gram factorizations $M_{\text{prime}} = L L^\dagger \succeq 0$. We prove the 4-Fold Equivalence connecting RH, the Herglotz property of logarithmic derivatives, de Branges phase monotonicity, and canonical Weyl function representations. Furthermore, we dissect the exact technical barrier of the off-axis cross-pairing term $-\sum \widehat{v}(\rho)\overline{\widehat{v}(1-\overline{\rho})}$ and the arithmetic sign problem in the explicit formula. The selective breakdown of this geometric structure in Epstein zeta functions is demonstrated.
\end{abstract}

\section{Introduction}
The Riemann Hypothesis states that all non-trivial zeros of the Riemann zeta function $\zeta(s)$ lie on the critical line $\operatorname{Re}(s) = 1/2$. In 1952, André Weil \cite{Weil1952} proved that RH is strictly equivalent to the non-negativity of the Weil quadratic functional on the admissible algebra of test functions:
\begin{equation}
\mathrm{RH} \iff Q_W(g) \ge 0 \quad \forall g \in \mathcal{T}_{\text{Weil}}.
\end{equation}

Recent developments by Connes--Consani--Moscovici \cite{CCM2025}, Suzuki \cite{Suzuki2026}, and Groskin \cite{Groskin2026a, Groskin2026b} have introduced continuous operator-theoretic and canonical system models for $Q_W$. The purpose of this paper is to provide a complete, transparent exposition of this framework, distinguishing rigorously established theorems from open analytical boundaries.

\section{The Functional-Analytic Framework}
Let $\mathcal{T}_{\text{Weil}}$ be the space of smooth test functions $g(x)$ satisfying the decay and pole-neutralization conditions $\widehat{g}(0) = 0, \widehat{g}(\pm i/2) = 0$. The Weil quadratic form associated with the explicit formula is given by:
\begin{equation}
Q_W(v) = W(v * \widetilde{v}), \quad \widetilde{v}(x) = \overline{v(-x)},
\end{equation}
where the functional $W(f)$ decomposes as:
\begin{equation}
W(f) = W_{\text{arch}}(f) + W_{\text{pole}}(f) - \sum_{n=1}^\infty \frac{\Lambda(n)}{\sqrt{n}} \left( f(\log n) + f(-\log n) \right).
\end{equation}

\section{Algebraic Factorization and Continuous Form Domains}

\subsection{Lemma 1: Gram Matrix Structure of Prime Modes}
\begin{lemma}
Let $\mathcal{D} = C_c^\infty(\mathbb{R})$. For any discrete Galerkin truncation of positive-type test functions $g_v = v * \widetilde{v}$, the arithmetic quadratic kernel is positive semi-definite (PSD) and factors as:
\begin{equation}
M_{\text{prime}} = \sum_{q \ge 2} \frac{\Lambda(q)}{\sqrt{q}} u_q u_q^\dagger = L L^\dagger \succeq 0,
\end{equation}
where $u_q = (q^{i\omega_1/2\pi}, \dots, q^{i\omega_N/2\pi})^T \in \mathbb{C}^N$ and $\Lambda(q) \ge 0$.
\end{lemma}
\begin{remark}
In the global Weil functional $W(f)$, the arithmetic contribution enters with a negative sign ($-M_{\text{prime}} \preceq 0$), reflecting the energetic subtraction of prime modes from the Archimedean background.
\end{remark}

\subsection{Lemma 2: Continuous Screw-Kernel and Friedrichs Extension}
\begin{lemma}[Suzuki \cite{Suzuki2026}]
The localized Weil form $Q_W^a$ on $H_0^1(-a, a)$ admits the continuous representation:
\begin{equation}
Q_W^a(v) = \iint_{(-a,a)^2} g(x-y) v'(x) \overline{v'(y)} \, dx dy,
\end{equation}
where $g(t) = \frac{1}{2}|t|\log|t| + A|t| + \sum_{n \le e^{|t|}} \frac{\Lambda(n)}{\sqrt{n}}(|t| - \log n) + r(t)$ is continuous. The associated operator $A_a = \operatorname{Friedrichs}(D^* G_a D)$ is self-adjoint and lower-semibounded ($Q_W^a \ge -C_a \|v\|^2$).
\end{lemma}

\section{The 4-Fold Equivalence Theorem}

\begin{theorem}[4-Fold Equivalence]
The following four analytical and arithmetic statements are strictly equivalent:
\begin{enumerate}
    \item \textbf{Riemann Hypothesis}: All non-trivial zeros satisfy $\operatorname{Re}(\rho) = 1/2$.
    \item \textbf{Herglotz Logarithmic Derivative}: The quotient $\operatorname{Im}\left( \frac{\xi'(1/2 - iz)}{\xi(1/2 - iz)} \right) > 0$ for all $z \in \mathbb{C}^+$.
    \item \textbf{de Branges Phase Monotonicity}: The phase function $\Phi(t) = -\arg\left(\xi(1/2-it) - i\xi'(1/2-it)\right)$ satisfies $\Phi'(t) > 0$ strictly for all $t \in \mathbb{R}$.
    \item \textbf{Suzuki Canonical Model Identification}: The limiting Weyl function $m_\infty(z) = \mathcal{M}\left[ z^2 \frac{\xi(1/2 - iz)}{\xi'(1/2 - iz)} \right]$ belongs to the Nevanlinna class $\mathcal{N}_0$.
\end{enumerate}
\end{theorem}

\section{Analysis of the Off-Axis Cross-Pairing Obstacle}
When decomposing the explicit formula over the zero set $Z_\zeta$, the zero-sum quadratic contribution on $v * \widetilde{v}$ is given by:
\begin{equation}
W_{\text{zeros}}(v * \widetilde{v}) = -\sum_{\rho \in Z_\zeta} \widehat{v}(\rho) \overline{\widehat{v}(1 - \overline{\rho})}.
\end{equation}

\begin{enumerate}
    \item \textbf{On-Axis Realization ($\beta = 1/2$)}: Since $1 - \overline{\rho} = \rho$, the pairing collapses to $-|\widehat{v}(\rho)|^2 \le 0$, matching the classical modulus-square structure.
    \item \textbf{Hypothetical Off-Axis Realization ($\beta = 1/2 + \delta, \delta \ne 0$)}: The pairing $\widehat{v}(\rho) \overline{\widehat{v}(1 - \overline{\rho})}$ becomes an off-diagonal bilinear form with differing exponential weightings. Establishing an unconditional lower bound $Q_W(v) \ge 0$ without a priori assuming $\delta = 0$ requires balancing this cross-pairing against the continuous screw kernel $g(x-y)$.
\end{enumerate}

\section{Epstein Zeta Counterexample Breakdown}
For Epstein zeta functions $Z_E(s)$ of binary quadratic forms with class number $h > 1$, off-critical zeros $\rho_E = 1/2 + \delta_E + i\gamma_E$ ($\delta_E > 0$) exist. In this setting:
\begin{enumerate}
    \item The pseudo-arithmetic coefficients $a_q$ take negative values; the Gram decomposition $M = L L^\dagger$ fails due to imaginary entries $\sqrt{a_q} \in i\mathbb{R}$.
    \item The Weyl-Titchmarsh function $m_a^E(z)$ develops poles in the open upper half-plane $\mathbb{C}^+$, degenerating into the generalized Nevanlinna class $\mathcal{N}_\kappa$ with negative index $\kappa \ge 1$.
    \item The underlying space degenerates into a Krein space, confirming that our framework selectively excludes non-Euler systems.
\end{enumerate}
